- [ ] **Step 2.1: 备份现状并 append custom_rules block**

```bash
cp kFresh/.swiftlint.yml kFresh/.swiftlint.yml.bak
cat >> kFresh/.swiftlint.yml <<'YAML'

# === v1.x-A: Kraftly 自定义规则（warning 级，仅建基线）===
custom_rules:
  no_silent_try_question_mark:
    name: "Silent try? in production code"
    regex: '\btry\?'
    message: "Use do/catch + Result or explicit error handling. try? silently swallows errors. If best-effort is intentional, add `// swiftlint:disable:next no_silent_try_question_mark` with a one-line DocC explanation."
    severity: warning
    match_kinds:
      - keyword

  no_hardcoded_design_tokens:
    name: "Hardcoded design tokens"
    regex: '(?:\.(?:system\(size:)|\.system\(weight:)|padding\(\d+(?:\.0)?\)|frame\(width: \d+(?:\.0)?, height: \d+(?:\.0)?\))'
    message: "Use AppFont / AppSpacing / WindowFrame tokens from kFoundation DesignSystem."
    severity: warning
    match_kinds:
      - identifier
      - keyword

  no_kuninstall_string:
    name: "kUninstall literal in source"
    regex: '"kUninstall"|kUninstall/|app\.kraftly\.kuninstall'
    message: "kFresh rename complete; remove all kUninstall literals from source."
    severity: warning
    match_kinds:
      - string
      - identifier

  no_unchecked_sendable:
    name: "@unchecked Sendable"
    regex: '@unchecked\s+Sendable'
    message: "@unchecked Sendable requires explicit justification; see CLAUDE.md §5.1."
    severity: warning
    match_kinds:
      - identifier
YAML
```

⚠️ 关键：spec 草稿的 `no_hardcoded_design_tokens` regex 首个 `\.` 只锚定第 1 个 alternation（标准正则 alternation 行为），其余 3 个分支可匹配文件任意位置。本计划用 `(?:\.\.|\.\.|\.\.)` 强制每支独立锚定。

```swift
import Foundation
import AppKit
import os

actor FDAuthorizer {
    private static let logger = Logger(subsystem: "app.kraftly.kfresh", category: "FDAuthorizer")
    private let fileManager = FileManager.default

    /// Check if FDA is granted by testing ~/Library accessibility
    func checkFDA() -> Bool {
        let testPath = fileManager.homeDirectoryForCurrentUser
            .appendingPathComponent("Library/Application Support")
        return fileManager.isReadableFile(atPath: testPath.path)
    }

    /// Open System Settings → Privacy → Full Disk Access
    nonisolated func requestFDA() {
        guard let url = URL(string: "x-apple.systempreferences:com.apple.preference.security?Privacy_AllFiles") else { return }
        NSWorkspace.shared.open(url)
    }

    /// Create a Security-Scoped Bookmark for persistent access
    func createBookmark(for url: URL) -> Data? {
        do {
            return try url.bookmarkData(options: .withSecurityScope,
                                        includingResourceValuesForKeys: nil,
                                        relativeTo: nil)
        } catch {
            Self.logger.error("createBookmark failed for \(url.path): \(error.localizedDescription)")
            return nil
        }
    }

    /// Resolve a Security-Scoped Bookmark
    func resolveBookmark(_ data: Data) -> URL? {
        var isStale = false
        do {
            let url = try URL(resolvingBookmarkData: data,
                              options: .withSecurityScope,
                              relativeTo: nil,
                              bookmarkDataIsStale: &isStale)
            _ = url.startAccessingSecurityScopedResource()
            return url
        } catch {
            Self.logger.error("resolveBookmark failed: \(error.localizedDescription)")
            return nil
        }
    }
}
```

- [ ] **Step 4.3: 重构 CaskParser**

```swift
    /// Run `pattern` once over `source` and return the first capture group, if any.
    private static func extract(_ source: String, pattern: String) -> String? {
        guard let regex = makeRegex(pattern),
              let match = regex.firstMatch(in: source, range: NSRange(source.startIndex..., in: source)),
              let range = Range(match.range(at: 1), in: source) else { return nil }
        return String(source[range])
    }

    /// Extract every `"..."` literal inside `key: [ ... ]` (e.g. `trash: ["a", "b"]`).
    /// `(?s)` enables single-line mode so `.` matches newlines, letting the array
    /// body span multiple lines (the typical Homebrew Cask formatting).
    private static func extractArray(_ source: String, key: String) -> [String] {
        let pattern = "(?s)\(key):\\s*\\[(.*?)\\]"
        guard let body = extract(source, pattern: pattern),
              let regex = makeRegex(#""([^"]+)""#) else { return [] }
        let range = NSRange(body.startIndex..., in: body)
        return regex.matches(in: body, range: range).compactMap {
            Range($0.range(at: 1), in: body).map { String(body[$0]) }
        }
    }

    /// Compile a regex or return nil for an invalid pattern. Centralised so the
    /// 2 call sites do not each need a `try?`.
    private static func makeRegex(_ pattern: String) -> NSRegularExpression? {
        // swiftlint:disable:next no_silent_try_question_mark
        // Pattern source is hardcoded by callers (no user input); invalid pattern
        // is a programmer error, but makeRegex returns nil rather than crashing
        // so a future static `let` typo does not take down the whole CaskParser.
        try? NSRegularExpression(pattern: pattern)
    }
```
